% Ajustes globales para los bloques de código resaltados por Pandoc.
% Las líneas conservan un tamaño legible y se ajustan al ancho disponible. El
% salto es exclusivamente visual: no introduce caracteres ni modifica la
% fuente QMD. Los tramos RTL se aíslan antes mediante un filtro de Pandoc.
\usepackage{fvextra}

% Pandoc declara Highlighting después de procesar include-in-header. Aplicar
% esta personalización antes quedaba sobrescrito en el documento final; por
% eso se difiere hasta \begin{document}.
\AtBeginDocument{%
  \RecustomVerbatimEnvironment{Highlighting}{Verbatim}{
    commandchars=\\\{\},
    fontsize=\small,
    tabsize=4,
    breaklines=true,
    breakanywhere=true,
    breaknonspaceingroup=true,
    breakpreferspaces=true,
    breakanywheresymbolpre={},
    breakanywheresymbolpost={},
    breaksymbolleft={},
    breaksymbolright={},
    showspaces=false,
    showtabs=false
  }%
}
